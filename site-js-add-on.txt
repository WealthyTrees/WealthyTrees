/* Daily Edition menu link: paste this at the very BOTTOM of site.js (after the last line). It adds "Edition" to the menu on every page. */
(function () {
  var n = document.querySelector("header nav");
  if (!n || n.querySelector('a[href="/edition.html"]')) return;
  var a = document.createElement("a"); a.href = "/edition.html"; a.textContent = "Edition";
  var m = n.querySelector('a[href="/markets.html"]'); n.insertBefore(a, m || n.querySelector(".navcta"));
  var s = document.createElement("style");
  s.textContent = "@media(min-width:641px){header .wrap{max-width:76rem}header nav a{padding:.7rem 1.1rem}}";
  document.head.appendChild(s);
})();
